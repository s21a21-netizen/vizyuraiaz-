function convertToMakeCodePython(midi) { const speed = parseFloat(speedInput.value) || 1.0; // すべてのトラックからノート（音符）情報を収集 let allNotes = []; midi.tracks.forEach(track => { track.notes.forEach(note => { allNotes.push({ midi: note.midi, time: note.time / speed, duration: note.duration / speed }); }); }); if (allNotes.length === 0) { output.value = "# MIDIデータから音符が検出できませんでした。"; copyBtn.style.display = "none"; return; } // 時間順にソート allNotes.sort((a, b) => a.time - b.time); let freqs = []; let durations = []; let currentTime = 0; allNotes.forEach(note => { // 直前の音との間に隙間（休符）があれば追加 const gap = note.time - currentTime; if (gap > 0.05) { // 50ms以上のスキップなら休符とする freqs.push(0); // 0Hz = 休符 durations.push(Math.round(gap * 1000)); } // 音符を追加（ミリ秒単位） freqs.push(midiToFreq(note.midi)); durations.push(Math.round(note.duration * 1000)); currentTime = note.time + note.duration; }); // MakeCode micro:bit V2 Python用コードの組み立て const pythonCode = `# MakeCode micro:bit Python用音階演奏プログラム # 変換元: ${midi.name || 'MIDI File'} notes = [${freqs.join(', ')}] durations = [${durations.join(', ')}] for i in range(len(notes)): freq = notes[i] duration = durations[i] if freq > 0: music.play(music.tone_playable(freq, duration), music.PlaybackMode.UNTIL_DONE) else: basic.pause(duration) `; output.value = pythonCode; copyBtn.style.display = "inline-block"; }
